#!/bin/bash
# ==================================================================================
# UNIFIED Pedagogy Manual PDF Export Utility
# Eliminates console noise, silences Chromium GCM/API logs, and forces clean print
# ==================================================================================
set -e

# Define directories
PROJECT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
MANUALS_DIR="$PROJECT_DIR/output_builds"
OUT_DIR="$PROJECT_DIR/output_pdfs"

mkdir -p "$OUT_DIR"

# Step 1: Automatically rebuild HTML blueprints to ensure fonts and centering are fresh
echo "============================================================"
echo "Step 1: Rebuilding HTML blueprints..."
echo "============================================================"
python3 "$PROJECT_DIR/build_manuals.py"

# List of files to process
HTML_FILES=(
  "01_couture_dame_tailleur_dames"
  "02_dossier_technique"
  "03_gradation"
  "04_moulage"
  "05_ost"
  "06_technologie_machines"
  "07_technologie_textile"
  "08_protection_consommateur"
  "09_entrepreneuriat"
)

echo ""
echo "============================================================"
echo "Step 2: Converting HTML to PDF with silenced logs..."
echo "============================================================"

# Check for Chrome
if command -v google-chrome &>/dev/null; then
    CHROME_CMD="google-chrome"
elif command -v chromium-browser &>/dev/null; then
    CHROME_CMD="chromium-browser"
elif command -v chromium &>/dev/null; then
    CHROME_CMD="chromium"
else
    echo "Error: Google Chrome or Chromium is required but was not found."
    exit 1
fi

for name in "${HTML_FILES[@]}"; do
    if [ -f "$MANUALS_DIR/$name.html" ]; then
        echo ">>> Converting: $name.html"
        
        # We redirect both stdout and stderr (2>&1 >/dev/null) and pass --disable-logging and --log-level=3
        # to ensure no GCM registration or delegate error noise reaches the console.
        $CHROME_CMD --headless \
                    --no-sandbox \
                    --disable-gpu \
                    --disable-logging \
                    --log-level=3 \
                    --print-to-pdf-no-header \
                    --print-to-pdf="$OUT_DIR/$name.pdf" \
                    "$MANUALS_DIR/$name.html" >/dev/null 2>&1
                    
        if [ -f "$OUT_DIR/$name.pdf" ]; then
            size=$(du -h "$OUT_DIR/$name.pdf" | cut -f1)
            echo "    ✓ Successfully generated: $name.pdf ($size)"
        else
            echo "    ✗ Failed to verify PDF output for $name"
        fi
    else
        echo "    ! Warning: Source file $name.html not found in $MANUALS_DIR"
    fi
done

# Also process registry files if present
if ls "$MANUALS_DIR"/*_registry.html &>/dev/null; then
    echo ""
    echo "============================================================"
    echo "Step 3: Processing Registry files..."
    echo "============================================================"
    for html_file in "$MANUALS_DIR"/*_registry.html; do
        base=$(basename "$html_file" .html)
        echo ">>> Converting registry: $base.html"
        $CHROME_CMD --headless \
                    --no-sandbox \
                    --disable-gpu \
                    --disable-logging \
                    --log-level=3 \
                    --print-to-pdf-no-header \
                    --print-to-pdf="$OUT_DIR/$base.pdf" \
                    "$html_file" >/dev/null 2>&1
        if [ -f "$OUT_DIR/$base.pdf" ]; then
            size=$(du -h "$OUT_DIR/$base.pdf" | cut -f1)
            echo "    ✓ Successfully generated: $base.pdf ($size)"
        fi
    done
fi

echo ""
echo "============================================================"
echo "Compilation Complete. PDF files are saved in: $OUT_DIR"
echo "============================================================"
